Service entrypoint backend

The shape of a Nova service: an effect fn main with explicit capabilities — IO and network arrive as parameters, never ambient authority.

// main.nova — a service that fetches, transforms, stores.
effect net fn fetchUsers(net: Net) -> Result[List[String], String] {
  let body = fetch("https://api.example.com/users", net)?
  return Ok(parseUsers(body))
}

effect fn main(io: IO, net: Net) {
  match fetchUsers(net) {
    Ok(users) => {
      writeFile("users.txt", join(users, "\n"), io)
      print("stored " + str(len(users)) + " users")
    },
    Err(e) => print("failed: " + e)
  }
}

Counter component frontend

React's component model, Java's types, none of the hook rules. State is compiler-managed and order-independent.

component Counter(title: String) {
  state count: Int = 0

  view {
    <div>
      <h1>{title}</h1>
      <p>{count}</p>
      <button onClick={count = count + 1}>Increment</button>
    </div>
  }
}

effect fn main() {
  let app = render(Counter(title: "Clicks"))
  app.fire("onClick")
  app.fire("onClick")
}

ai fn summarize(text: String) -> Summary
  model = "nova-mock-1"
  temperature = 0.3

effect fn main() {

  print("score: " + str(s.score))

Streams pub/sub platform

Kafka-style topics, consumer groups and offsets — in the standard library. Same code runs on the in-memory provider locally and real Kafka in production.

stream events: String

effect fn drain(sub: Subscription[String], count: Int) -> Int {
  return match sub.poll() {
    None => count,
    Some(msg) => {
      print(msg.value)
      sub.commit(msg)
      drain(sub, count + 1)
    }
  }
}

effect fn main() {
  let sub = events.subscribe(group: "workers")
  events.publish("hello")
  events.publish(key: "k1", value: "world")
  print("drained: " + str(drain(sub, 0)))
}
main()

Result error handling errors

Failure in the signature, propagation with ?, exhaustive handling with match. No exception tax, no silent undefined.

effect fn fetchUser(id: Int) -> Result[String, String] {
  if id == 1 { return Ok("Ada") }
  return Err("user " + str(id) + " not found")
}

effect fn greetUser(id: Int) -> Result[String, String] {
  let name = fetchUser(id)?   // Err returns early from greetUser
  return Ok("Hello, " + name + "!")
}

effect fn main() {
  match greetUser(1) {
    Ok(msg) => print(msg),
    Err(e)  => print("Error: " + e)
  }
  match greetUser(7) {
    Ok(msg) => print(msg),
    Err(e)  => print("Error: " + e)
  }
}

Java POJO → Nova migration

The poster child of nova migrate: forty lines of getters, setters, equals and hashCode become one declaration. See the full mapping tables in the migration guide.

Before — Java
public class Point {
  private double x;
  private double y;
  public Point(double x, double y) {
    this.x = x; this.y = y;
  }
  public double getX() { return x; }
  public double getY() { return y; }
  // + equals + hashCode + toString
  //   (~40 lines total)
}
After — Nova
data class Point(x: Float, y: Float)

let p = Point(x: 1.0, y: 2.0)
print(p)                          // Point(x: 1.0, y: 2.0)
print(p == Point(x: 1.0, y: 2.0)) // true

React useState → Nova migration

Hooks become declarations. There is nothing to memorize about call order because there is no call order.

Before — React
function Counter({ title }) {
  const [count, setCount] = useState(0);
  // don't call me conditionally.
  // don't call me in a loop. good luck.
  return (
    <div>
      <h1>{title}</h1>
      <p>{count}</p>
      <button onClick={() => setCount(c => c + 1)}>
        Increment
      </button>
    </div>
  );
}
After — Nova
component Counter(title: String) {
  state count: Int = 0   // declare anywhere, in any order

  view {
    <div>
      <h1>{title}</h1>
      <p>{count}</p>
      <button onClick={count = count + 1}>
        Increment
      </button>
    </div>
  }
}
